cmake_minimum_required(VERSION 3.29)
project(ChargeReplenishment VERSION 1.0.0 LANGUAGES CXX RC)
find_package(D2RLPlugin 0.4.0 EXACT CONFIG REQUIRED)
add_library(charge-replenishment SHARED src/plugin.cpp src/native.cpp src/plugin.rc)
set_source_files_properties(src/plugin.rc PROPERTIES COMPILE_DEFINITIONS
    "CHARGE_REPLENISHMENT_CONFIG_PATH=\"${CMAKE_CURRENT_SOURCE_DIR}/charge-replenishment.toml\"")
target_link_libraries(charge-replenishment PRIVATE D2RLPlugin::D2RLPlugin)
target_compile_features(charge-replenishment PRIVATE cxx_std_20)
target_compile_definitions(charge-replenishment PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(charge-replenishment PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/permissive-;/utf-8;/Brepro>")
set_target_properties(charge-replenishment PROPERTIES PREFIX "d2rl-" MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(CHARGE_REPLENISHMENT_TESTS "Build recharge policy and SDK integration tests" OFF)
if(CHARGE_REPLENISHMENT_TESTS)
    enable_testing()
    foreach(test_name IN ITEMS policy runtime native_stats)
        add_executable(${test_name}_tests tests/${test_name}_tests.cpp)
        target_link_libraries(${test_name}_tests PRIVATE D2RLPlugin::D2RLPlugin)
        target_compile_features(${test_name}_tests PRIVATE cxx_std_20)
        target_compile_definitions(${test_name}_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_compile_options(${test_name}_tests PRIVATE /W4 /WX /permissive- /utf-8)
        set_target_properties(${test_name}_tests PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded)
        add_test(NAME ${test_name} COMMAND ${test_name}_tests)
    endforeach()
endif()
